\newcommand{\safeImage}[2][10mm]{%
  \IfFileExists{#2}{%
    \includegraphics[width=#1]{#2}%
  }{%
    \begin{tikzpicture}[baseline=(current bounding box.center)]
      \draw[red, very thick] (0,0) rectangle (#1, #1);
      \draw[red, very thick] (0,0) -- (#1, #1);
      \draw[red, very thick] (#1,0) -- (0,#1);
    \end{tikzpicture}%
  }%
}
