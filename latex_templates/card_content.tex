\newcommand{\safeImage}[2][10mm]{%
  \IfFileExists{#2}{%
    \includegraphics[width=#1]{#2}%
  }{%
    \begin{tikzpicture}[baseline=(current bounding box.center)]
      \draw[red, very thick] (0,0) rectangle (#1, #1);
      \draw[red, very thick] (0,0) -- (#1, #1);
      \draw[red, very thick] (#1,0) -- (0,#1);
    \end{tikzpicture}%
  }%
}%

% Circular profile picture macro
\newcommand{\profile}[1][23mm]{%
  \IfFileExists{\profileImage}{%
    \begin{tikzpicture}
      \clip (0,0) circle[radius=#1/2];
      \node[inner sep=0pt, outer sep=0pt, anchor=center] at (0,0) {\includegraphics[width=#1]{\profileImage}};
    \end{tikzpicture}%
  }{%
    \begin{tikzpicture}
      \draw[teal!30, thick] (0,0) circle[radius=#1/2];
      \node[text=teal!50, font=\small] at (0,0) {Foto};
    \end{tikzpicture}%
  }%
}
